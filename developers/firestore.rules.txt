rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // المدير: بريدك المؤكَّد (غيّره هنا إن كنت تدخل ببريد آخر)
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email == 'alfahamnet@gmail.com';
    }

    // ملف المطور: يقرؤه ويكتبه صاحبه، ويقرؤه المدير
    match /devProfiles/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
      allow read: if isAdmin();
    }

    // طلبات النشر
    match /submissions/{id} {
      // المطور ينشئ طلبه بحالة pending فقط
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == 'pending';
      // المطور يقرأ طلباته، والمدير يقرأ الكل
      allow read: if request.auth != null && (resource.data.uid == request.auth.uid || isAdmin());
      // المدير فقط يغيّر الحالة (قبول/رفض) أو يحذف
      allow update, delete: if isAdmin();

      // الصور (الأيقونة ولقطات الشاشة والبانر)
      match /media/{mid} {
        allow create: if request.auth != null
          && getAfter(/databases/$(database)/documents/submissions/$(id)).data.uid == request.auth.uid;
        allow read: if request.auth != null
          && (get(/databases/$(database)/documents/submissions/$(id)).data.uid == request.auth.uid || isAdmin());
        allow delete: if isAdmin();
        allow update: if false;
      }
    }
  }
}
