rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ملف المطور: يقرؤه ويكتبه صاحبه، وتقرؤه لوحة التحكم
    match /devProfiles/{uid} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // حجز أسماء المطورين: مستند واحد لكل اسم، فلا يتكرر الاسم بين مطورين
    match /devNames/{key} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
      allow update, delete: if request.auth != null && resource.data.uid == request.auth.uid;
    }

    // حجز أسماء الحزم: يُنشأ مرة واحدة عند إرسال تطبيق جديد، وتحذفه لوحة التحكم
    match /packages/{pkg} {
      allow get: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
      allow delete: if true;
    }

    // طلبات النشر
    match /submissions/{id} {
      // طلب تعديل تطبيق منشور
      function isUpdate() {
        return request.resource.data.get('type', 'new') == 'update';
      }
      // لا يعدّل التطبيق إلا صاحبه: إما مسجّل باسمه في packages، أو له طلب مقبول بنفس الحزمة
      function ownsPackage() {
        let pkg = request.resource.data.packageName.lower();
        let base = request.resource.data.get('baseSubmissionId', '');
        let viaReg = exists(/databases/$(database)/documents/packages/$(pkg))
          && get(/databases/$(database)/documents/packages/$(pkg)).data.uid == request.auth.uid;
        let viaSub = base is string && base.size() > 0
          && exists(/databases/$(database)/documents/submissions/$(base))
          && get(/databases/$(database)/documents/submissions/$(base)).data.uid == request.auth.uid
          && get(/databases/$(database)/documents/submissions/$(base)).data.status == 'approved'
          && get(/databases/$(database)/documents/submissions/$(base)).data.packageName.lower() == pkg;
        return viaReg || viaSub;
      }

      // المطور ينشئ طلبه بحالة pending فقط (والتعديل لصاحب الحزمة فقط)
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == 'pending'
        && (!isUpdate() || ownsPackage());
      // لوحة التحكم تفتح بلا تسجيل دخول: القراءة والقبول والرفض والحذف مفتوحة
      allow read, update, delete: if true;

      // الصور (الأيقونة ولقطات الشاشة والبانر)
      match /media/{mid} {
        allow create: if request.auth != null
          && getAfter(/databases/$(database)/documents/submissions/$(id)).data.uid == request.auth.uid;
        allow read, delete: if true;
        allow update: if false;
      }
    }
  }
}
