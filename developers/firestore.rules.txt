rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // المدير: بريدك المؤكَّد (غيّره هنا إن كنت تدخل ببريد آخر)
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email == 'alfahamnet@gmail.com';
    }

    // ملف المطور: يقرؤه ويكتبه صاحبه، ويقرؤه المدير
    match /devProfiles/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
      allow read: if isAdmin();
    }

    // حجز أسماء المطورين: مستند واحد لكل اسم، فلا يتكرر الاسم بين مطورين
    match /devNames/{key} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
      allow update, delete: if request.auth != null && resource.data.uid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // حجز أسماء الحزم: يُنشأ مرة واحدة عند إرسال تطبيق جديد ولا يتغير بعدها
    match /packages/{pkg} {
      allow get: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // طلبات النشر
    match /submissions/{id} {
      // طلب تعديل تطبيق منشور
      function isUpdate() {
        return request.resource.data.get('type', 'new') == 'update';
      }
      // لا يعدّل التطبيق إلا صاحب الحزمة المسجّلة باسمه
      function ownsPackage() {
        let pkg = request.resource.data.packageName.lower();
        return exists(/databases/$(database)/documents/packages/$(pkg))
          && get(/databases/$(database)/documents/packages/$(pkg)).data.uid == request.auth.uid;
      }

      // المطور ينشئ طلبه بحالة pending فقط (والتعديل لصاحب الحزمة فقط)
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == 'pending'
        && (!isUpdate() || ownsPackage());
      // المطور يقرأ طلباته، والمدير يقرأ الكل
      allow read: if request.auth != null && (resource.data.uid == request.auth.uid || isAdmin());
      // المدير فقط يغيّر الحالة (قبول/رفض) أو يحذف
      allow update, delete: if isAdmin();

      // الصور (الأيقونة ولقطات الشاشة والبانر)
      match /media/{mid} {
        allow create: if request.auth != null
          && getAfter(/databases/$(database)/documents/submissions/$(id)).data.uid == request.auth.uid;
        allow read: if request.auth != null
          && (get(/databases/$(database)/documents/submissions/$(id)).data.uid == request.auth.uid || isAdmin());
        allow delete: if isAdmin();
        allow update: if false;
      }
    }
  }
}
